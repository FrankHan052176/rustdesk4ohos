cmake_minimum_required(VERSION 3.20)
project(rd_windows_nvenc LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "rd_windows_nvenc is Windows-only")
endif()
add_library(rd_windows_nvenc STATIC windows_nvenc.cpp)
target_compile_features(rd_windows_nvenc PRIVATE cxx_std_17)
target_include_directories(rd_windows_nvenc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(rd_windows_nvenc PRIVATE RD_WINDOWS_NVENC_BUILD WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(rd_windows_nvenc PRIVATE d3d11 dxgi)
if(MSVC)
  target_compile_options(rd_windows_nvenc PRIVATE /W4 /EHsc /permissive-)
endif()
